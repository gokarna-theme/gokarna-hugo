{{- /* Fingerprint of what search-index.json is built from, added to its URL so that browsers and CDNs fetch a fresh index once posts or search settings change. Call with partialCached, keyed by language, so it is computed once per build. */ -}} {{- $cfg := site.Params.search | default dict -}} {{- $sig := slice hugo.Version (jsonify $cfg) -}} {{- range partial "search-pages.html" . -}} {{- $sig = $sig | append .RelPermalink .Title (.Date.Format "2006-01-02") (jsonify (partial "page-tags.html" .)) (md5 .Content) -}} {{- end -}} {{- return substr (md5 (delimit $sig "\n")) 0 10 -}}
